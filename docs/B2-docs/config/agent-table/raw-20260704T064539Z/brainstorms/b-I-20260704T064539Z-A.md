```python
def acp_session_connect(project, raw_key, role, agent_id, slot,
                        message=None, force_new=False, parent_role="manager"):
    """统一 session 操作选择。active detection + fallback safety net.

    Returns: (session_id, operation_used)
      operation_used: 'prompt' | 'resume' | 'fork' | 'new'
    """
    base_tool = slot.split('-')[0] if '-' in slot else slot

    # ── Active detection ──
    if force_new:
        return _do_new(project, raw_key, role, agent_id, slot, base_tool,
                       message, reason="force_new")

    existing = agent_list_by_raw_key(project, raw_key)
    if existing and existing.get(f"{role}_id"):
        old_sid = _extract_session_id(existing, role)
        if old_sid:
            is_alive = _bridge_ping(project, base_tool, old_sid)
            if is_alive:
                if message:
                    _bridge_rpc(project, base_tool, "session/prompt", {...})
                    return old_sid, "prompt"
                return old_sid, "return"
            else:
                try:
                    _bridge_rpc(project, base_tool, "session/resume", {...})
                    if message:
                        _bridge_rpc(project, base_tool, "session/prompt", {...})
                    acp_session_register(project, raw_key, role, agent_id, old_sid, slot)
                    return old_sid, "resume"
                except Exception as e:
                    log_warn(f"resume failed for {role}/{agent_id}: {e}, falling back to new")
                    return _do_new(project, raw_key, role, agent_id, slot, base_tool,
                                   message, reason=f"resume failed: {e}")

    # ── No session → fork or new ──
    mgr_slot = _get_manager_slot(project)
    mgr_base = mgr_slot.split('-')[0] if mgr_slot else ""
    if base_tool == mgr_base:
        try:
            mgr_sid = agent_get_session(project, "__manager__", "manager")
            if mgr_sid:
                sid = _bridge_rpc(project, base_tool, "session/fork",
                                  {"sessionId": mgr_sid, "role": role, "agentId": agent_id})
                acp_session_register(project, raw_key, role, agent_id, sid, slot)
                return sid, "fork"
        except Exception as e:
            log_warn(f"fork failed for {role}/{agent_id}: {e}, falling back to new")
    else:
        log_info(f"cross-tool ({base_tool} vs {mgr_base}), using new")

    return _do_new(project, raw_key, role, agent_id, slot, base_tool,
                   message, reason="no-session")

def _do_new(project, raw_key, role, agent_id, slot, base_tool, message=None, reason=""):
    sid = _bridge_rpc(project, base_tool, "session/new",
                      {"role": role, "agentId": agent_id})
    acp_session_register(project, raw_key, role, agent_id, sid, slot)
    if reason:
        log_info(f"session/new for {role}/{agent_id}: {reason}")
    return sid, "new"
```
