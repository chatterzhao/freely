# B2 Brainstorm: multi-bridge implementation

## 1. socket/pid per-tool

```bash
_acp_bridge_pid_file() {
  local project="$1" base_tool="${2:-}"
  echo "/tmp/freely-acp-$(_acp_bridge_short_hash "$project")-${base_tool}.pid"
}
_acp_bridge_socket() {
  local project="$1" base_tool="${2:-}"
  echo "/tmp/freely-acp-$(_acp_bridge_short_hash "$project")-${base_tool}.sock"
}
```

## 2. daemon start: scan + start all bridges

local _unique_tools=()
for slot in $("$BIN/agent-table-query" --list-slots 2>/dev/null); do
    local bt="${slot%%-*}"
    [[ " ${_unique_tools[*]} " == *" $bt "* ]] && continue

done

## 5. _resolve_bridge_socket in agent_working.py

```python
def _resolve_bridge_socket(project, base_tool=""):
    h = hashlib.sha256(os.path.abspath(project).encode()).hexdigest()[:16]
    return f"/tmp/freely-acp-{h}-{base_tool}.sock"
```

## 6. _stop_acp_bridge parameterized

```bash
_stop_acp_bridge() {
  local project="$1" base_tool="${2:-}"
  if [[ -n "$base_tool" ]]; then
    # Stop specific bridge
    pid_file="$(_acp_bridge_pid_file "$project" "$base_tool")"
    socket="$(_acp_bridge_socket "$project" "$base_tool")"
    _kill_bridge "$pid_file" "$socket"
  else
    # Stop all bridges
    for pid_f in /tmp/freely-acp-*-*.pid; do
        [[ -f "$pid_f" ]] || continue
        local pid; pid=$(cat "$pid_f" 2>/dev/null)
        [[ -n "$pid" ]] && kill "$pid" 2>/dev/null
        rm -f "$pid_f" "${pid_f%.pid}.sock"
    done
  fi
}
```
